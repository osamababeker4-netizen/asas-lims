-- 1. الفروع (تم تعديل بريدة إلى الرس)

  name_ar text not null,
  city text not null,
  address text,
  lat double precision,
  lng double precision,
  is_active boolean default true,
  created_at timestamp default now()
);

-- 2. المستخدمون والصلاحيات (إصلاح الدخول)
create table profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  username text unique not null,
  full_name text,
  role text check (role in ('admin','manager','lab_technician','field_engineer','quality','client')) default 'lab_technician',
  branch_id uuid references branches(id),
  avatar_url text,
  is_active boolean default true,
  created_at timestamp default now()
);

-- RLS for profiles
alter table profiles enable row level security;
create policy "public read profiles" on profiles for select using (true);
create policy "users update own" on profiles for update using (auth.uid() = id);

-- 3. العملاء
create table clients (
  id uuid primary key default uuid_generate_v4(),
  name text not null,
  phone text,
  email text,
  company text,
  created_at timestamp default now()
);

-- 4. المشاريع
create table projects (
  id uuid primary key default uuid_generate_v4(),
  project_code text unique not null,
  name text not null,
  owner_name text,
  location_text text,
  branch_id uuid references branches(id),
  client_id uuid references clients(id),
  priority text check (priority in ('low','medium','high','urgent')) default 'medium',
  status text check (status in ('planning','field','lab','reporting','done','on_hold')) default 'planning',
  progress int default 0 check (progress >=0 and progress <=100),
  start_date date,
  due_date date,
  created_by uuid references profiles(id),
  created_at timestamp default now()
);

-- 5. أوامر العمل
create table work_orders (
  id uuid primary key default uuid_generate_v4(),
  wo_number text unique not null,
  project_id uuid references projects(id) on delete cascade,
  title text not null,
  assigned_to uuid references profiles(id),
  priority text default 'medium',
  due_date date,
  status text default 'open' check (status in ('open','in_progress','review','closed')),
  created_at timestamp default now()
);

-- 6. الزيارات الميدانية والبرنامج الميداني
create table field_visits (
  id uuid primary key default uuid_generate_v4(),
  project_id uuid references projects(id) on delete cascade,
  work_order_id uuid references work_orders(id),
  license_number text,
  contractor_name text,
  sector text,
  layer_number text,
  latitude double precision,
  longitude double precision,
  field_notes text,
  visit_date timestamp default now(),
  created_by uuid references profiles(id)
);

-- 7. العينات
create table samples (
  id uuid primary key default uuid_generate_v4(),
  sample_code text unique not null,
  project_id uuid references projects(id) on delete cascade,
  field_visit_id uuid references field_visits(id),
  material_type text, -- تربة, خرسانة, اسفلت, ركام, حديد
  official_plan text,
  received_at timestamp default now(),
  status text check (status in ('received','in_lab','testing','completed','rejected')) default 'received',
  barcode text,
  qr_code text
);

-- 8. دليل الاختبارات (ASTM)
create table test_catalog (
  id uuid primary key default uuid_generate_v4(),
  code text unique not null, -- C39, D6926
  name_ar text not null,
  name_en text,
  category text, -- خرسانة, اسفلت, تربة...
  standard text, -- ASTM C39
  version text,
  worksheet_template_url text,
  created_at timestamp default now()
);

-- 9. الاختبارات المنفذة
create table tests (
  id uuid primary key default uuid_generate_v4(),
  test_number text unique not null,
  sample_id uuid references samples(id) on delete cascade,
  catalog_id uuid references test_catalog(id),
  technician_id uuid references profiles(id),
  result_value text,
  result_unit text,
  status text check (status in ('pending','in_progress','under_review','approved','rejected')) default 'pending',
  tested_at timestamp default now()
);

-- 10. الأجهزة والمعايرة
create table devices (
  id uuid primary key default uuid_generate_v4(),
  name text not null,
  serial_number text unique,
  manufacturer text,
  model text,
  last_calibration date,
  next_calibration date,
  certificate_url text,
  notes text,
  status text generated always as (
    case
      when next_calibration < current_date then 'expired'
      when next_calibration <= current_date + interval '30 days' then 'warning'
      else 'valid'
    end
  ) stored,
  branch_id uuid references branches(id),
  created_at timestamp default now()
);

-- 11. التقارير
create table reports (
  id uuid primary key default uuid_generate_v4(),
  report_number text unique not null,
  sample_id uuid references samples(id),
  test_id uuid references tests(id),
  project_id uuid references projects(id),
  status text check (status in ('draft','review','approved','issued')) default 'draft',
  pdf_url text,
  qr_verification_code text,
  approved_by uuid references profiles(id),
  created_at timestamp default now()
);

-- 12. وثائق الجودة QMS
create table quality_docs (
  id uuid primary key default uuid_generate_v4(),
  doc_code text not null, -- QMS-00
  title text not null,
  category text check (category in ('QMS-00_devices','QMS-01_procedures','QMS-02_worksheets','QMS-03_admin','proficiency','personnel')),
  version text,
  status text default 'active',
  file_url text,
  created_at timestamp default now()
);

-- 13. مسودات واتساب
create table whatsapp_drafts (
  id uuid primary key default uuid_generate_v4(),
  destination text, -- مجتمع أساس
  reference_type text, -- project, report, device
  reference_id uuid,
  draft_text text not null,
  status text default 'pending_review' check (status in ('pending_review','approved','sent','rejected')),
  created_at timestamp default now()
);

-- 14. سجل التدقيق
create table audit_logs (
  id uuid primary key default uuid_generate_v4(),
  user_id uuid references profiles(id),
  action text not null,
  entity_type text,
  entity_id uuid,
  details jsonb,
  created_at timestamp default now()
);

-- Indexes for performance
create index idx_projects_branch on projects(branch_id);
create index idx_samples_project on samples(project_id);
create index idx_tests_sample on tests(sample_id);
create index idx_devices_next_cal on devices(next_calibration);
create index idx_reports_status on reports(status);

-- Function for auto barcode
create or replace function generate_sample_code() returns trigger as $$
begin
  if NEW.sample_code is null then
    NEW.sample_code := 'ASAS-' || to_char(now(),'YYMM') || '-' || substr(uuid_generate_v4()::text,1,6);
  end if;
  return NEW;
end;
$$ language plpgsql;

create trigger trg_sample_code before insert on samples for each row execute function generate_sample_code();
